// ═══════════════════════════════════════════════════════
// 第一部分：配置区（按需修改）

const dataFolder = "生产数据";
const teams = ["甲班", "乙班", "丙班", "丁班"];
const processBaseScore = 40;
const waterStandard = 0; // 0=自动判断，1=123标准，2=考核标准

// ═══════════════════════════════════════════════════════
// 第二部分：工具函数
// ═══════════════════════════════════════════════════════

function cleanNumber(v) {
  if (v === undefined || v === null) return null;
  if (typeof v === "number") return v;
  if (typeof v !== "string") return null;
  let s = v.trim()
    .replace(/^["'‘’"'"'"]+|["'‘’"'"'"]+$/g, "")
    .replace(/,/g, "")
    .replace(/，/g, "");
  let m = s.match(/^-?\d+(\.\d+)?/);
  if (m) {
    let n = parseFloat(m[0]);
    return isNaN(n) ? null : n;
  }
  return null;
}

function safeNum(p, key) {
  let v = cleanNumber(p.value(key));
  return v !== null ? v : 0;
}

function hasAnyData(p) {
  return ["蒸汽消耗", "糖浆加量", "水消耗", "电消耗"]
    .some(k => cleanNumber(p.value(k)) !== null);
}

function formatNum(v) {
  if (v === undefined || v === null || v === "" || v === "-") return "-";
  let n = parseFloat(v);
  if (isNaN(n)) return String(v);
  return n.toFixed(4).replace(/\.?0+$/, "");
}

function fmtDate(v) {
  if (v === undefined || v === null) return "-";
  if (typeof v === "object" && typeof v.toFormat === "function") return v.toFormat("yyyy-MM-dd");
  return String(v);
}

// ═══════════════════════════════════════════════════════
// 第三部分：水分处理（两套标准）
// ═══════════════════════════════════════════════════════

function get123Standard(val) {
  if (val === 2) return ["👌", 0];
  if (val < 2) return ["🍂", -5];
  return ["💦", -10];
}

function getExamStandard(val) {
  if (val >= 11 && val <= 12) return ["👌", 0];
  if (val < 11) return ["🍂", -5];
  return ["💦", -10];
}

function getWaterStatus(val) {
  if (waterStandard === 1) return get123Standard(val);
  if (waterStandard === 2) return getExamStandard(val);
  if (val <= 5) return get123Standard(val);
  return getExamStandard(val);
}

function formatWater(m) {
  let val = cleanNumber(m);
  if (val === null) return "-";
  return getWaterStatus(val)[0];
}

function calcWaterScore(m) {
  let val = cleanNumber(m);
  if (val === null) return 0;
  return getWaterStatus(val)[1];
}

// ═══════════════════════════════════════════════════════
// 第四部分：排名算法
// ═══════════════════════════════════════════════════════

function rank(arr, key, asc = true) {
  const sorted = [...arr].sort((a, b) => {
    const av = a[1][key], bv = b[1][key];
    const aok = typeof av === "number", bok = typeof bv === "number";
    if (!aok && !bok) return 0;
    if (!aok) return 1;
    if (!bok) return -1;
    return asc ? av - bv : bv - av;
  });
  const vals = sorted.filter(x => typeof x[1][key] === "number").map(x => x[1][key]);
  const unique = [...new Set(vals)].sort((a, b) => asc ? a - b : b - a);
  const map = {};
  unique.forEach((v, i) => map[v] = i + 1);
  return map;
}

function Table(props) {
  const head = props.head;
  const rows = props.rows;
  return (
    <table>
      <thead>
        <tr>{head.map((col, idx) => <th key={idx}>{col}</th>)}</tr>
      </thead>
      <tbody>
        {rows.map((row, rIdx) => (
          <tr key={rIdx}>{row.map((cell, cIdx) => <td key={cIdx}>{cell}</td>)}</tr>
        ))}
      </tbody>
    </table>
  );
}

// ═══════════════════════════════════════════════════════
// 第六部分：主视图

return function View() {
  // 数据读取：路径 + 必须有班组字段
  const pages = dc.useQuery('@page and path("生产数据")')
    .filter(p => p.value("班组") != null);

  // ── 班组统计 ──
  const teamStats = {};
  const allRepairs = [];

  teams.forEach(team => {
    const rows = pages
      .filter(p => p.value("班组") === team)
      .sort((a, b) => String(fmtDate(a.value("日期"))).localeCompare(String(fmtDate(b.value("日期")))));

    if (rows.length === 0) return;

    let 蒸汽合计 = 0, 糖浆合计 = 0, 水合计 = 0, 电合计 = 0;
    let 正常班数 = 0, 水分扣分 = 0, 合格数 = 0, 检维修数 = 0;
    const 明细 = [];

    rows.forEach(p => {
      const isRepair = p.value("类型") === "检维修";
      const hasData = hasAnyData(p) && !isRepair;

      if (hasData) {
        蒸汽合计 += safeNum(p, "蒸汽消耗");
        糖浆合计 += safeNum(p, "糖浆加量");
        水合计   += safeNum(p, "水消耗");
        电合计   += safeNum(p, "电消耗");
        正常班数 += 1;
      }
      if (isRepair) 检维修数 += 1;

      const 本条扣分 = calcWaterScore(p.value("水分"));
      水分扣分 += 本条扣分;
      if (hasData && 本条扣分 === 0) 合格数 += 1;

      if (isRepair) {
        allRepairs.push({ 班组: team, 日期: fmtDate(p.value("日期")) });
        明细.push({
          日期: fmtDate(p.value("日期")), 类型: "🛠",
          蒸汽: "-", 糖浆: "-", 水: "-", 电: "-",
          水分: formatWater(p.value("水分"))
        });
      } else {
        明细.push({
          日期: fmtDate(p.value("日期")), 类型: "✅",
          蒸汽: formatNum(p.value("蒸汽消耗")),
          糖浆: formatNum(p.value("糖浆加量")),
          水: formatNum(p.value("水消耗")),
          电: formatNum(p.value("电消耗")),
          水分: formatWater(p.value("水分"))
        });
      }
    });

    teamStats[team] = {
      蒸汽: 蒸汽合计, 糖浆: 糖浆合计, 水: 水合计, 电: 电合计,
      扣分: 水分扣分, 班数: 正常班数, 检维修数: 检维修数, 合格数: 合格数,
      比: 糖浆合计 ? parseFloat((蒸汽合计 / 糖浆合计).toFixed(4)) : "-",
      水均: 正常班数 ? parseFloat((水合计 / 正常班数).toFixed(4)) : "-",
      电均: 正常班数 ? parseFloat((电合计 / 正常班数).toFixed(4)) : "-",
      工艺分: 正常班数 ? parseFloat((processBaseScore / 正常班数 * 合格数).toFixed(4)) : 0,
      明细: 明细
    };
  });

  // ── 排名 ──
  const statsList = teams.map(t => [t, teamStats[t]]).filter(x => x[1]);
  const rRatio = rank(statsList, "比", true);
  const rWater = rank(statsList, "水均", true);
  const rElec  = rank(statsList, "电均", true);

  statsList.forEach(([t, s]) => {
    s.排名比 = rRatio[s.比] || "-";
    s.排名水 = rWater[s.水均] || "-";
    s.排名电 = rElec[s.电均] || "-";
    s.积分 = (typeof s.排名比 === "number" ? s.排名比 : 0) +
             (typeof s.排名水 === "number" ? s.排名水 : 0) +
             (typeof s.排名电 === "number" ? s.排名电 : 0);
  });
  const rScore = rank(statsList, "积分", true);
  statsList.forEach(([t, s]) => { s.总排名 = rScore[s.积分] || "-"; });

  // ── 渲染 ──
  return (
    <div>
      <h2>📖 图例说明</h2>
      <Table head={["符号", "含义"]} rows={[
        ["✅", "正常班"], ["🛠", "检维修"],
        ["👌", "水分合格"], ["🍂", "水分偏干（扣5分）"], ["💦", "水分偏潮（扣10分）"]
      ]} />

      <h2>📊 消耗统计汇总</h2>
      <Table head={["班组","蒸汽用量","糖浆加量","水量","电量","水分扣分"]}
        rows={statsList.map(([t, s]) => [t, formatNum(s.蒸汽), formatNum(s.糖浆), formatNum(s.水), formatNum(s.电), s.扣分])} />

      <h2>📈 平均分</h2>
      <Table head={["班组","蒸汽÷糖浆","水量÷正常班","电量÷正常班","工艺分"]}
        rows={statsList.map(([t, s]) => [t, formatNum(s.比), formatNum(s.水均), formatNum(s.电均), formatNum(s.工艺分)])} />

      <h2>🏆 积分排名</h2>
      <Table head={["班组","蒸汽÷糖浆=排名","水消耗排名","电消耗排名","各班积分","最低消耗排名"]}
        rows={statsList.map(([t, s]) => [t, s.排名比, s.排名水, s.排名电, s.积分, s.总排名])} />

      {allRepairs.length > 0 && (
        <div>
          <h2>🔧 检维修记录</h2>
          <Table head={["班组","日期"]} rows={allRepairs.map(r => [r.班组, r.日期])} />
        </div>
      )}

      {teams.map(team => {
        const s = teamStats[team];
        if (!s) return null;
        const total = s.班数 + s.检维修数;
        return (
          <div key={team}>
            <h3>📋 {team} 明细（{total}）</h3>
            <Table head={["日期","类型","蒸汽消耗","糖浆加量","水消耗","电消耗","水分"]}
              rows={s.明细.map(d => [d.日期, d.类型, d.蒸汽, d.糖浆, d.水, d.电, d.水分]).concat([
                [<strong>小计</strong>, "✅:" + s.班数 + " | 🛠:" + s.检维修数,
                 formatNum(s.蒸汽), formatNum(s.糖浆), formatNum(s.水), formatNum(s.电), s.扣分]
              ])} />
          </div>
        );
      })}
    </div>
  );
}
